// ==UserScript==
// @name         AnimeOn NYA Achievement Full Logger
// @namespace    animeon-nya-debug
// @version      1.0
// @description  Captures achievement/network/DOM/storage activity and searches for NYA-XXXX-XXXX
// @match        https://animeon.cc/*
// @run-at       document-start
// @grant        none
// ==/UserScript==

(() => {
    'use strict';
    if (window.__NYA_LOGGER__?.running) {
        console.warn('NYA Logger уже установлен');
        return { running: true };
    }
    try { window.__NYA_LOGGER__?.stop?.(); } catch {}

    const VERSION = '1.0';

    const NYA_RE = /\bNYA-[0123456789ABCDEFGHJKMNPQRSTVWXYZ]{4}-[0123456789ABCDEFGHJKMNPQRSTVWXYZ]{4}\b/gi;
    const INTERESTING_RE =
        /NYA-|frame_token|achievement|achievements|unlock|notification|terminal|purr|morse/i;

    const MAX_BODY = 2 * 1024 * 1024;
    const MAX_DOM_TEXT = 20000;
    const PANEL_POSITION_KEY = 'animeon_nya_logger_position_v1';

    const started = new Date();
    const logs = [];
    const foundCodes = new Set();

    let seq = 0;
    let stopped = false;
    let domObserver = null;
    let performanceObserver = null;
    const restores = [];

    /* ---------------------------------------------------------
       UTILITIES
       --------------------------------------------------------- */

    function now() {
        return new Date().toISOString();
    }

    function safeString(value) {
        try {
            if (typeof value === 'string') return value;

            if (value instanceof URLSearchParams)
                return value.toString();

            if (value instanceof FormData) {
                const obj = {};

                for (const [k, v] of value.entries()) {
                    obj[k] =
                        typeof v === 'string'
                            ? v
                            : `[File ${v?.name || ''}]`;
                }

                return JSON.stringify(obj);
            }

            if (value instanceof Blob)
                return `[Blob type=${value.type} size=${value.size}]`;

            if (value instanceof ArrayBuffer)
                return `[ArrayBuffer ${value.byteLength}]`;

            if (ArrayBuffer.isView(value))
                return `[${value.constructor.name} ${value.byteLength}]`;

            return JSON.stringify(value);
        } catch {
            try {
                return String(value);
            } catch {
                return '[unserializable]';
            }
        }
    }

    function limit(text, max = MAX_BODY) {
        text = safeString(text);

        if (text.length <= max) return text;

        return (
            text.slice(0, max) +
            `\n\n[TRUNCATED: original length ${text.length}]`
        );
    }

    function sanitizeHeaders(headers) {
        const out = {};

        try {
            if (!headers) return out;

            const h = new Headers(headers);

            for (const [key, value] of h.entries()) {
                if (
                    /authorization|cookie|set-cookie|token|jwt|session|csrf/i.test(
                        key
                    )
                ) {
                    out[key] = '[REDACTED]';
                } else {
                    out[key] = value;
                }
            }
        } catch {}

        return out;
    }

    function extractCodes(value, source = '') {
        const text = safeString(value);

        if (!text) return [];

        const matches = text.match(NYA_RE) || [];

        for (const raw of matches) {
            const code = raw.toUpperCase();

            if (!foundCodes.has(code)) {
                foundCodes.add(code);

                console.log(
                    '%c[NYA LOGGER] CODE FOUND:',
                    'background:#6d28d9;color:white;font-size:16px;padding:4px 8px',
                    code,
                    source
                );

                addLog('NYA_FOUND', {
                    code,
                    source
                });

                showFound(code);
            }
        }

        return matches;
    }

    function addLog(type, data = {}) {
        if (stopped) return null;
        const entry = {
            id: ++seq,
            time: now(),
            type,
            ...data
        };

        logs.push(entry);

        try {
            extractCodes(data, type);
        } catch {}

        return entry;
    }

    function relevant(text) {
        try {
            return INTERESTING_RE.test(safeString(text));
        } catch {
            return false;
        }
    }

    function bodySummary(text) {
        const value = limit(text);

        extractCodes(value, 'BODY');

        return value;
    }

    /* ---------------------------------------------------------
       FETCH
       --------------------------------------------------------- */

    const nativeFetch = window.fetch;

    window.fetch = async function (...args) {
        const requestId = ++seq;

        let url = '';
        let method = 'GET';
        let requestHeaders = {};
        let requestBody = null;

        try {
            const input = args[0];
            const init = args[1] || {};

            if (input instanceof Request) {
                url = input.url;
                method = init.method || input.method || 'GET';

                requestHeaders = sanitizeHeaders(
                    init.headers || input.headers
                );

                try {
                    if (
                        method !== 'GET' &&
                        method !== 'HEAD'
                    ) {
                        const clone = input.clone();
                        requestBody = bodySummary(await clone.text());
                    }
                } catch {}
            } else {
                url = String(input);
                method = init.method || 'GET';
                requestHeaders = sanitizeHeaders(init.headers);

                if (init.body != null)
                    requestBody = bodySummary(init.body);
            }
        } catch {}

        addLog('FETCH_REQUEST', {
            requestId,
            method,
            url,
            headers: requestHeaders,
            body: requestBody
        });

        const start = performance.now();

        try {
            const response = await nativeFetch.apply(this, args);

            let responseBody = null;
            let contentType = '';

            try {
                contentType =
                    response.headers.get('content-type') || '';

                const clone = response.clone();

                if (
                    /json|text|javascript|xml|html|plain/i.test(
                        contentType
                    )
                ) {
                    responseBody = bodySummary(await clone.text());
                } else {
                    const blob = await clone.blob();

                    responseBody =
                        `[Binary response type=${blob.type} size=${blob.size}]`;
                }
            } catch (e) {
                responseBody =
                    `[Unable to read response: ${e}]`;
            }

            addLog('FETCH_RESPONSE', {
                requestId,
                method,
                url,
                status: response.status,
                statusText: response.statusText,
                durationMs: Math.round(
                    (performance.now() - start) * 100
                ) / 100,
                headers: sanitizeHeaders(response.headers),
                body: responseBody
            });

            if (
                relevant(url) ||
                relevant(responseBody)
            ) {
                console.log(
                    '[NYA LOGGER][FETCH]',
                    method,
                    response.status,
                    url,
                    responseBody
                );
            }

            return response;
        } catch (error) {
            addLog('FETCH_ERROR', {
                requestId,
                method,
                url,
                durationMs: Math.round(
                    (performance.now() - start) * 100
                ) / 100,
                error: String(error)
            });

            throw error;
        }
    };

    /* ---------------------------------------------------------
       XMLHttpRequest
       --------------------------------------------------------- */

    const nativeOpen = XMLHttpRequest.prototype.open;
    const nativeSend = XMLHttpRequest.prototype.send;
    const nativeSetHeader =
        XMLHttpRequest.prototype.setRequestHeader;

    XMLHttpRequest.prototype.open = function (
        method,
        url,
        ...rest
    ) {
        this.__nya = {
            id: ++seq,
            method: String(method),
            url: String(url),
            headers: {},
            start: 0
        };

        return nativeOpen.call(this, method, url, ...rest);
    };

    XMLHttpRequest.prototype.setRequestHeader = function (
        key,
        value
    ) {
        try {
            if (this.__nya) {
                this.__nya.headers[key] =
                    /authorization|cookie|token|session|csrf/i.test(
                        key
                    )
                        ? '[REDACTED]'
                        : String(value);
            }
        } catch {}

        return nativeSetHeader.call(this, key, value);
    };

    XMLHttpRequest.prototype.send = function (body) {
        const info =
            this.__nya ||
            {
                id: ++seq,
                method: '?',
                url: '?',
                headers: {}
            };

        info.start = performance.now();

        addLog('XHR_REQUEST', {
            requestId: info.id,
            method: info.method,
            url: info.url,
            headers: info.headers,
            body:
                body == null
                    ? null
                    : bodySummary(body)
        });

        this.addEventListener('loadend', () => {
            let responseBody = null;

            try {
                if (
                    this.responseType === '' ||
                    this.responseType === 'text'
                ) {
                    responseBody =
                        bodySummary(this.responseText);
                } else if (
                    this.responseType === 'json'
                ) {
                    responseBody =
                        bodySummary(this.response);
                } else {
                    responseBody =
                        `[XHR responseType=${this.responseType}]`;
                }
            } catch {}

            addLog('XHR_RESPONSE', {
                requestId: info.id,
                method: info.method,
                url: info.url,
                status: this.status,
                statusText: this.statusText,
                durationMs: Math.round(
                    (performance.now() - info.start) * 100
                ) / 100,
                body: responseBody
            });

            if (
                relevant(info.url) ||
                relevant(responseBody)
            ) {
                console.log(
                    '[NYA LOGGER][XHR]',
                    info.method,
                    this.status,
                    info.url,
                    responseBody
                );
            }
        });

        return nativeSend.call(this, body);
    };

    /* ---------------------------------------------------------
       WEBSOCKET
       --------------------------------------------------------- */

    const NativeWebSocket = window.WebSocket;

    window.WebSocket = class extends NativeWebSocket {
        constructor(...args) {
            super(...args);

            const wsId = ++seq;

            addLog('WS_OPENING', {
                wsId,
                url: String(args[0])
            });

            this.addEventListener('open', () => {
                addLog('WS_OPEN', {
                    wsId,
                    url: this.url
                });
            });

            this.addEventListener('message', event => {
                const data = bodySummary(event.data);

                addLog('WS_MESSAGE_IN', {
                    wsId,
                    url: this.url,
                    data
                });

                if (relevant(data))
                    console.log(
                        '[NYA LOGGER][WS IN]',
                        this.url,
                        data
                    );
            });

            this.addEventListener('close', event => {
                addLog('WS_CLOSE', {
                    wsId,
                    url: this.url,
                    code: event.code,
                    reason: event.reason
                });
            });

            this.addEventListener('error', () => {
                addLog('WS_ERROR', {
                    wsId,
                    url: this.url
                });
            });
        }

        send(data) {
            const text = bodySummary(data);

            addLog('WS_MESSAGE_OUT', {
                url: this.url,
                data: text
            });

            if (relevant(text))
                console.log(
                    '[NYA LOGGER][WS OUT]',
                    this.url,
                    text
                );

            return super.send(data);
        }
    };

    /* ---------------------------------------------------------
       CUSTOM EVENTS
       --------------------------------------------------------- */

    const nativeDispatch =
        EventTarget.prototype.dispatchEvent;

    EventTarget.prototype.dispatchEvent = function (event) {
        try {
            let detail = null;

            if ('detail' in event)
                detail = bodySummary(event.detail);

            if (
                relevant(event.type) ||
                relevant(detail)
            ) {
                addLog('EVENT_DISPATCH', {
                    eventType: event.type,
                    detail
                });

                console.log(
                    '[NYA LOGGER][EVENT]',
                    event.type,
                    event.detail
                );
            }
        } catch {}

        return nativeDispatch.call(this, event);
    };

    /* ---------------------------------------------------------
       STORAGE
       --------------------------------------------------------- */

    function hookStorage(name, storage) {
        if (!storage) return;

        try {
            const originalSet =
                storage.setItem.bind(storage);
            const originalRemove =
                storage.removeItem.bind(storage);
            const originalClear =
                storage.clear.bind(storage);

            storage.setItem = function (key, value) {
                const k = String(key);
                const v = bodySummary(value);

                if (
                    relevant(k) ||
                    relevant(v)
                ) {
                    addLog(`${name}_SET`, {
                        key: k,
                        value: v
                    });

                    console.log(
                        `[NYA LOGGER][${name}]`,
                        k,
                        value
                    );
                }

                return originalSet(key, value);
            };

            storage.removeItem = function (key) {
                if (relevant(key))
                    addLog(`${name}_REMOVE`, {
                        key: String(key)
                    });

                return originalRemove(key);
            };

            storage.clear = function () {
                addLog(`${name}_CLEAR`);

                return originalClear();
            };
            restores.push(() => {
                storage.setItem = originalSet;
                storage.removeItem = originalRemove;
                storage.clear = originalClear;
            });
        } catch (e) {
            addLog('STORAGE_HOOK_ERROR', {
                storage: name,
                error: String(e)
            });
        }
    }

    try {
        hookStorage('LOCALSTORAGE', window.localStorage);
    } catch {}

    try {
        hookStorage(
            'SESSIONSTORAGE',
            window.sessionStorage
        );
    } catch {}

    /* ---------------------------------------------------------
       CLIPBOARD
       --------------------------------------------------------- */

    try {
        if (
            navigator.clipboard &&
            navigator.clipboard.writeText
        ) {
            const originalWriteText =
                navigator.clipboard.writeText.bind(
                    navigator.clipboard
                );

            navigator.clipboard.writeText =
                async function (text) {
                    addLog('CLIPBOARD_WRITE', {
                        text: bodySummary(text)
                    });

                    return originalWriteText(text);
                };
            restores.push(() => { navigator.clipboard.writeText = originalWriteText; });
        }
    } catch {}

    /* ---------------------------------------------------------
       DOM OBSERVER
       --------------------------------------------------------- */

    function scanDOMNode(node, reason) {
        try {
            let text = '';

            if (node.nodeType === Node.TEXT_NODE)
                text = node.nodeValue || '';
            else if (node instanceof Element)
                text =
                    node.innerText ||
                    node.textContent ||
                    '';

            if (!text) return;

            text = text.slice(0, MAX_DOM_TEXT);

            const codes = extractCodes(
                text,
                `DOM:${reason}`
            );

            if (
                codes.length ||
                relevant(text)
            ) {
                addLog('DOM_MATCH', {
                    reason,
                    tag:
                        node.nodeType === Node.ELEMENT_NODE
                            ? node.tagName
                            : '#text',
                    text: limit(text, MAX_DOM_TEXT)
                });

                console.log(
                    '[NYA LOGGER][DOM]',
                    reason,
                    text
                );
            }
        } catch {}
    }

    function startObserver() {
        if (!document.documentElement) {
            setTimeout(startObserver, 25);
            return;
        }

            domObserver = new MutationObserver(
            mutations => {
                for (const m of mutations) {
                    if (m.type === 'characterData') {
                        scanDOMNode(
                            m.target,
                            'characterData'
                        );
                    }

                    for (const node of m.addedNodes)
                        scanDOMNode(node, 'added');
                }
            }
        );

            domObserver.observe(document.documentElement, {
            subtree: true,
            childList: true,
            characterData: true
        });

        addLog('DOM_OBSERVER_STARTED');
    }

    startObserver();

    /* ---------------------------------------------------------
       PERFORMANCE RESOURCE OBSERVER
       --------------------------------------------------------- */

    try {
        performanceObserver = new PerformanceObserver(list => {
            for (const entry of list.getEntries()) {
                if (relevant(entry.name)) {
                    addLog('PERFORMANCE_RESOURCE', {
                        name: entry.name,
                        initiatorType:
                            entry.initiatorType,
                        duration:
                            entry.duration,
                        transferSize:
                            entry.transferSize
                    });
                }
            }
        });

        performanceObserver.observe({
            type: 'resource',
            buffered: true
        });
    } catch {}

    /* ---------------------------------------------------------
       EXPORT
       --------------------------------------------------------- */

    function exportLog() {
        const result = {
            logger: 'AnimeOn NYA Achievement Full Logger',
            version: VERSION,
            started: started.toISOString(),
            exported: now(),
            page: location.href,

            nyaFormat:
                'NYA-XXXX-XXXX',

            knownAlphabet:
                '0123456789ABCDEFGHJKMNPQRSTVWXYZ',

            foundCodes:
                [...foundCodes],

            totalEntries:
                logs.length,

            logs
        };

        const blob = new Blob(
            [JSON.stringify(result, null, 2)],
            {
                type: 'application/json;charset=utf-8'
            }
        );

        const url =
            URL.createObjectURL(blob);

        const a =
            document.createElement('a');

        a.href = url;

        a.download =
            `animeon_nya_capture_${new Date()
                .toISOString()
                .replace(/[:.]/g, '-')}.json`;

        document.documentElement.appendChild(a);

        a.click();

        a.remove();

        setTimeout(
            () => URL.revokeObjectURL(url),
            5000
        );

        console.log(
            `[NYA LOGGER] Exported ${logs.length} entries; NYA codes:`,
            [...foundCodes]
        );
    }

    /* ---------------------------------------------------------
       SMALL STATUS PANEL
       --------------------------------------------------------- */

    function createPanel() {
        if (stopped) return;
        if (!document.body) {
            setTimeout(createPanel, 100);
            return;
        }

        if (
            document.getElementById(
                '__nya_logger_panel'
            )
        )
            return;

        const panel =
            document.createElement('div');

        panel.id = '__nya_logger_panel';

        Object.assign(panel.style, {
            position: 'fixed', right: '20px', top: '90px', zIndex: '2147483647',
            width: '205px', overflow: 'hidden', color: '#e4e4e7',
            background: 'linear-gradient(145deg,rgba(18,18,24,.98),rgba(10,10,14,.98))',
            border: '1px solid rgba(255,255,255,.13)', borderRadius: '12px',
            font: '11px/1.45 system-ui,-apple-system,"Segoe UI",sans-serif',
            boxShadow: '0 14px 34px rgba(0,0,0,.48),inset 0 1px rgba(255,255,255,.04)'
        });

        panel.innerHTML = `
            <div id="__nya_logger_header" style="display:flex;align-items:center;justify-content:space-between;padding:9px 10px;border-bottom:1px solid rgba(255,255,255,.08);cursor:grab;user-select:none">
                <div style="display:flex;align-items:center;gap:7px;font-weight:700;color:#f4f4f5"><span style="width:7px;height:7px;border-radius:50%;background:#a855f7;box-shadow:0 0 9px #a855f7"></span>Nya Logger</div>
                <button id="__nya_logger_export" title="Скачать журнал найденных кодов" aria-label="Скачать журнал найденных кодов" style="display:grid;place-items:center;width:26px;height:26px;border:1px solid rgba(255,255,255,.14);border-radius:7px;background:rgba(255,255,255,.04);color:#d4d4d8;cursor:pointer;line-height:1"><svg width="14" height="14" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="M12 3v12"/><path d="m7 10 5 5 5-5"/><path d="M5 21h14"/></svg></button>
            </div>
            <div style="padding:9px 10px">
                <div style="display:flex;justify-content:space-between;color:#a1a1aa"><span>Коды</span><strong id="__nya_logger_count" style="font:700 16px/1.1 ui-monospace,SFMono-Regular,monospace;color:#d8b4fe">0</strong></div>
                <div id="__nya_logger_status" style="margin-top:7px;padding-top:7px;border-top:1px solid rgba(255,255,255,.08);color:#71717a">Наблюдение активно</div>
            </div>`;

        document.body.appendChild(panel);

        try {
            const saved = JSON.parse(localStorage.getItem(PANEL_POSITION_KEY));
            if (Number.isFinite(saved?.x) && Number.isFinite(saved?.y)) {
                panel.style.right = 'auto';
                panel.style.left = `${saved.x}px`;
                panel.style.top = `${saved.y}px`;
            }
        } catch {}

        const header = panel.querySelector('#__nya_logger_header');
        let dragging = false;
        let offsetX = 0;
        let offsetY = 0;

        header?.addEventListener('mousedown', event => {
            if (event.target.closest('button')) return;
            const rect = panel.getBoundingClientRect();
            dragging = true;
            offsetX = event.clientX - rect.left;
            offsetY = event.clientY - rect.top;
            panel.style.right = 'auto';
            header.style.cursor = 'grabbing';
            event.preventDefault();
        });

        const onMove = event => {
            if (!dragging) return;
            const x = Math.max(0, Math.min(window.innerWidth - panel.offsetWidth, event.clientX - offsetX));
            const y = Math.max(0, Math.min(window.innerHeight - panel.offsetHeight, event.clientY - offsetY));
            panel.style.left = `${x}px`;
            panel.style.top = `${y}px`;
        };
        const onUp = () => {
            if (!dragging) return;
            dragging = false;
            if (header) header.style.cursor = 'grab';
            const rect = panel.getBoundingClientRect();
            try { localStorage.setItem(PANEL_POSITION_KEY, JSON.stringify({ x: rect.left, y: rect.top })); } catch {}
        };
        document.addEventListener('mousemove', onMove);
        document.addEventListener('mouseup', onUp);
        restores.push(() => { document.removeEventListener('mousemove', onMove); document.removeEventListener('mouseup', onUp); });

        panel
            .querySelector(
                '#__nya_logger_export'
            )
            .addEventListener(
                'click',
                exportLog
            );
    }

    function showFound(code) {
        createPanel();

        setTimeout(() => {
            const status =
                document.getElementById(
                    '__nya_logger_status'
                );

            const count = document.getElementById('__nya_logger_count');
            if (count) count.textContent = String(foundCodes.size);
            if (status) {
                status.replaceChildren();
                const label = document.createElement('div');
                label.textContent = 'Найденные коды';
                label.style.cssText = 'margin-bottom:4px;color:#a1a1aa';
                const codes = document.createElement('div');
                codes.textContent = [...foundCodes].join(' · ');
                codes.style.cssText = 'color:#86efac;font:700 10px/1.55 ui-monospace,SFMono-Regular,monospace;word-break:break-word';
                status.append(label, codes);
            }
        }, 0);
    }

    const onDomReady = () => createPanel();
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', onDomReady, { once: true });
    } else {
        createPanel();
    }

    /* Ctrl + Shift + Y = EXPORT */

    const onExportShortcut = event => {
            if (
                event.ctrlKey &&
                event.shiftKey &&
                event.code === 'KeyY'
            ) {
                event.preventDefault();
                exportLog();
            }
        };
    window.addEventListener('keydown', onExportShortcut, true);

    /* ---------------------------------------------------------
       INITIAL SNAPSHOT
       --------------------------------------------------------- */

    addLog('LOGGER_STARTED', {
        version: VERSION,
        page: location.href,
        userAgent:
            navigator.userAgent
    });

    console.log(
        '%c[NYA LOGGER] ACTIVE',
        'background:#7c3aed;color:white;font-size:15px;padding:5px 10px',
        'Получай ачивку. После этого нажми EXPORT JSON или Ctrl+Shift+Y.'
    );

    /* Debug access */
    window.__NYA_LOGGER__ = {
        running: true,
        logs,
        foundCodes,
        export: exportLog,
        stop: () => {
            if (stopped) return;
            stopped = true;
            try { window.fetch = nativeFetch; } catch {}
            try { XMLHttpRequest.prototype.open = nativeOpen; XMLHttpRequest.prototype.send = nativeSend; XMLHttpRequest.prototype.setRequestHeader = nativeSetHeader; } catch {}
            try { window.WebSocket = NativeWebSocket; } catch {}
            try { EventTarget.prototype.dispatchEvent = nativeDispatch; } catch {}
            for (const restore of restores) { try { restore(); } catch {} }
            try { domObserver?.disconnect(); } catch {}
            try { performanceObserver?.disconnect(); } catch {}
            try { document.removeEventListener('DOMContentLoaded', onDomReady); } catch {}
            try { window.removeEventListener('keydown', onExportShortcut, true); } catch {}
            try { document.getElementById('__nya_logger_panel')?.remove(); } catch {}
            try { delete window.__NYA_LOGGER__; } catch {}
            console.log('[NYA LOGGER] остановлен');
        }
    };
})();
