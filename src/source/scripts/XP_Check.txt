    function getSlug() {
        const m = location.pathname.match(/^\/user\/([^/?#]+)/i);

        return m?.[1]
            ? decodeURIComponent(m[1]).toLowerCase()
            : String(TARGET_USER || '').trim().toLowerCase() || null;
    }

    // ---------------------------------------------------------
    // Перетаскивание
    // ---------------------------------------------------------

    const header =
        panel.querySelector('.xpmon-header');

    let dragging = false;
    let offsetX = 0;
    let offsetY = 0;

    header.addEventListener('mousedown', e => {

        if (e.target.closest('button')) {
            return;
        }

        dragging = true;

        const r =
            panel.getBoundingClientRect();

        offsetX =
            e.clientX - r.left;

        offsetY =
            e.clientY - r.top;

        panel.style.right = 'auto';

        e.preventDefault();
    });

    document.addEventListener('mousemove', e => {

        if (!dragging) {
            return;
        }

        let x =
            e.clientX - offsetX;

        let y =
            e.clientY - offsetY;

        x = Math.max(
            0,
            Math.min(
                window.innerWidth -
                panel.offsetWidth,
                x
            )
        );

        y = Math.max(
            0,
            Math.min(
                window.innerHeight -
                panel.offsetHeight,
                y
            )
        );

        panel.style.left =
            `${x}px`;

        panel.style.top =
            `${y}px`;
    });

    document.addEventListener('mouseup', () => {

        if (!dragging) {
            return;
        }

        dragging = false;

        const r =
            panel.getBoundingClientRect();

        localStorage.setItem(
            STORAGE_KEY,
            JSON.stringify({
                x: r.left,
                y: r.top
            })
        );
    });

    // Восстановление позиции

    try {
        const pos =
            JSON.parse(
                localStorage.getItem(
                    STORAGE_KEY
                )
            );

        if (
            pos &&
            Number.isFinite(pos.x) &&
            Number.isFinite(pos.y)
        ) {
            panel.style.right = 'auto';

            panel.style.left =
                `${pos.x}px`;

            panel.style.top =
                `${pos.y}px`;
        }

    } catch {}

    // ---------------------------------------------------------
    // Уровень доступен в меню аккаунта на каждой странице AnimeOn.
    // ---------------------------------------------------------

    function toFiniteNumber(value) {
        if (typeof value === 'number') return Number.isFinite(value) ? value : null;
        if (typeof value !== 'string') return null;

        const normalized = value
            .replace(/\u00a0/g, ' ')
            .replace(/\s+/g, '')
            .replace(',', '.');
        if (!normalized || !/^-?\d+(?:\.\d+)?$/.test(normalized)) return null;

        const number = Number(normalized);
        return Number.isFinite(number) ? number : null;
    }

    function readLevelFromPage() {
        const menu = document.querySelector('[aria-label*="Меню аккаунта"], [aria-label*="уровень"]');
        const menuText = `${menu?.getAttribute('aria-label') || ''} ${menu?.textContent || ''}`;
        const menuMatch = menuText.match(/уровень\s*[:\-]?\s*([\d\s\u00a0.,]+)/i);
        const menuLevel = toFiniteNumber(menuMatch?.[1]);
        if (menuLevel !== null) return menuLevel;

        // На странице /user/... это первое поле профиля. Берём только
        // самостоятельную строку, чтобы не спутать уровень другого пользователя.
        const pageMatch = String(document.body?.innerText || '').match(/(?:^|\n)\s*Уровень\s*[:\-]?\s*([\d\s\u00a0.,]+)\s*(?:\n|$)/i);
        return toFiniteNumber(pageMatch?.[1]);
    }

    // API не во всех версиях сайта кладёт уровень в profile.level:
    // он может лежать в stats/progress как строка или число. Ищем его в
    // объекте уже найденного пользователя, не в ответе целиком, чтобы не
    // спутать с уровнем аниме или достижения.
    function readLevelFromProfile(profile) {
        const keys = [
            'level',
            'current_level',
            'user_level',
            'experience_level',
            'xp_level',
            'lvl',
            'userLevel',
            'currentLevel',
            'experienceLevel',
            'xpLevel'
        ];
        const seen = new WeakSet();

        function walk(value, depth = 0) {
            if (!value || typeof value !== 'object' || depth > 5 || seen.has(value)) return null;
            seen.add(value);

            for (const key of keys) {
                const number = toFiniteNumber(value[key]);
                if (number !== null && number >= 0) return number;
            }

            for (const child of Object.values(value)) {
                const found = walk(child, depth + 1);
                if (found !== null) return found;
            }
            return null;
        }

        return walk(profile);
    }

    // /api/users/:slug отдает total_xp, но не обязан содержать уровень.
    // Сам профиль текущей сессии на разных версиях AnimeOn лежит в одном из
    // этих маршрутов; запрашиваем их редко, чтобы не нагружать сайт.
    async function readLevelFromCurrentSession() {
        if (cachedAccountLevel !== null) return cachedAccountLevel;
        if (Date.now() - lastLevelProbe < 60_000) return null;
        lastLevelProbe = Date.now();

        const paths = ['/api/users/me', '/api/auth/me', '/api/user/profile', '/api/profile'];
        for (const path of paths) {
            try {
                const response = await fetch(path, {
                    credentials: 'include', cache: 'no-store',
                    headers: { Accept: 'application/json' }
                });
                if (!response.ok) continue;
                const level = readLevelFromProfile(await response.json());
                if (level !== null) {
                    cachedAccountLevel = level;
                    return level;
                }
            } catch {}
        }
        return null;
    }

    // ---------------------------------------------------------
    // Извлечение объекта пользователя из JSON
    // ---------------------------------------------------------

    function findProfileObject(data, slug) {

        const seen =
            new WeakSet();

        let best = null;

        function walk(v) {

            if (
                !v ||
                typeof v !== 'object'
            ) {
                return;
            }

            if (seen.has(v)) {
                return;
            }

            seen.add(v);

            try {

                if (
                    Number.isFinite(v.total_xp) &&
                    String(
                        v.username_slug || ''
                    ).toLowerCase() === slug
                ) {
                    best = v;

                    return;
                }

            } catch {}

            for (
                const key of Object.keys(v)
            ) {

                if (best) {
                    return;
                }

                try {
                    walk(v[key]);
                } catch {}
            }
        }

        walk(data);

        return best;
    }

    // ---------------------------------------------------------
    // НАСТОЯЩИЙ API-запрос
    // ---------------------------------------------------------

    async function fetchProfile() {

        const slug =
            getSlug();

        if (!slug) {
            throw new Error(
                'Не удалось определить профиль'
            );
        }

        const response =
            await fetch(
                `/api/users/${encodeURIComponent(slug)}?t=${Date.now()}`,
                {
                    method: 'GET',

                    credentials: 'include',

                    cache: 'no-store',

                    headers: {
                        Accept:
                            'application/json'
                    }
                }
            );

        if (!response.ok) {

            throw new Error(
                `API HTTP ${response.status}`
            );
        }

        const data =
            await response.json();

        const profile =
            findProfileObject(
                data,
                slug
            );

        if (
            !profile ||
            !Number.isFinite(
                profile.total_xp
            )
        ) {
            throw new Error(
                'total_xp не найден'
            );
        }

        return {
            xp:
                Number(
                    profile.total_xp
                ),

            /*
             * Если API когда-либо начнёт
             * возвращать level прямо в объекте,
             * используем его.
             *
             * Сейчас резерв — уровень,
             * отображаемый на /user/...
             */
            level:
                readLevelFromProfile(profile) ??
                await readLevelFromCurrentSession() ??
                readLevelFromPage(),

            profile
        };
    }

    // ---------------------------------------------------------
    // Обновление
    // ---------------------------------------------------------

    async function refresh() {

        if (busy) {
            return;
        }

        busy = true;

        dotEl.className =
            'xpmon-dot loading';

        statusEl.textContent =
            'Запрос API…';

        refreshButton
            .classList
            .add('loading');

        try {

            const result =
                await fetchProfile();

            const xp =
                result.xp;

            xpEl.textContent =
                xp.toLocaleString(
                    'ru-RU'
                );

            if (
                Number.isFinite(
                    result.level
                )
            ) {
                levelEl.textContent =
                    result.level
                        .toLocaleString(
                            'ru-RU'
                        );
            }

            deltaEl.className =
                'delta-value';

            if (lastXP === null) {

                deltaEl.textContent =
                    '—';

            } else {

                const delta =
                    xp - lastXP;

                if (delta > 0) {

                    deltaEl.textContent =
                        '+' +
                        delta.toLocaleString(
                            'ru-RU'
                        );

                    deltaEl
                        .classList
                        .add('positive');

                } else if (delta < 0) {

                    deltaEl.textContent =
                        delta.toLocaleString(
                            'ru-RU'
                        );

                    deltaEl
                        .classList
                        .add('negative');

                } else {

                    deltaEl.textContent =
                        '0';
                }
            }

            lastXP = xp;

            dotEl.className =
                'xpmon-dot online';

            statusEl.textContent =
                `${getSlug()} · API`;

            updatedEl.textContent =
                'Обновлено: ' +
                new Date()
                    .toLocaleTimeString(
                        'ru-RU'
                    );

        } catch (error) {
            dotEl.className =
                'xpmon-dot error';

            statusEl.textContent =
                error.message;

        } finally {

            busy = false;

            refreshButton
                .classList
                .remove('loading');
        }
    }

    // ---------------------------------------------------------
    // Запуск
    // ---------------------------------------------------------

    refreshButton.addEventListener(
        'click',
        refresh
    );

    refresh();

    const timer = setInterval(refresh, POLL_MS);

    window.__ANIMEON_XP_MONITOR__ = {
        running: true,
        stop() {
            clearInterval(timer);
            try { panel.remove(); } catch {}
            try { style.remove(); } catch {}
            try { delete window.__ANIMEON_XP_MONITOR__; } catch {}
        }
    };

    return { running: true, targetUser: TARGET_USER };

})();
